import java.util.Random;

public class MyThread extends Thread{
    public MyThread(String name){
        super(name);
    }

    @Override
    public void run(){
        Random random=new Random();
        for(int i=0; i<5; i++){
            System.out.println(Thread.currentThread().getName()+" "+i);
            try {
                sleep(random.nextInt(1000)*10);
            } catch (InterruptedException e) {
                e.printStackTrace();
            }
        }
    }
}

public class MTThread extends Thread{
    private String name;
    private int table;

    public MTThread(String name,int table){
        super(name);
        this.table=table;
    }

    @Override
    public void run(){
        for(int i=0; i<10; i++){
            System.out.println(table + " * " + i +" = "+(table*i));
            try {
                sleep(1000);
            } catch (InterruptedException e) {
                e.printStackTrace();
            }
        }
    }
}

FThread.java

public class FThread extends Thread{
    private int input;

    public FThread(String name, int input){
        super(name);
        this.input=input;

    }

    private int findFactorial(int i){
        if(i==0 || i==1){
            return i;
        }
        else{
            return i*findFactorial(i-1);
        }
    }

    @Override
    public void run(){
        for(int i=1;i<=input; i++){
            System.out.println("Factorial of "+ i +" is "+ findFactorial(i));
        }
    }

}
