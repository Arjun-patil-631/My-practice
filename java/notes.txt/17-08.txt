===================singleton.java===============
----------------------------------------------
//this class is defined to allow instantiation exactly once-Singleton design patern
public class singleton{

    //static variable to tract the instance of this class
    private static singleton s=null;
    // to restrict the object creation
    private singleton(){

    }

    //static method to restrict object creation exactly once
    public static singleton createSingleton(){
        if(s==null){
            s=new singleton();
        }
        return s;
    }
}
---------------------------------------------------

=============singletonDemo.java===========
------------------------------------------
public class singletonDemo{
    public static void main(String[] args) {
        singleton s1=singleton.createSingleton();
        System.out.println(s1);
        singleton s2=singleton.createSingleton();
        System.out.println(s2);
    }
}
