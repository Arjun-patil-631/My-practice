a super class variable can can be used to point any subclass object 

public class A{
    public void MethodA(){
        System.out.println("From the method A of class A....");
    }
} 

public class B{
    public void MethodB(){
        System.out.println("From the method B of class B....");
    }
} 

public class C{
    public void MethodC(){
        System.out.println("From the method C of class C....");
    }
}
